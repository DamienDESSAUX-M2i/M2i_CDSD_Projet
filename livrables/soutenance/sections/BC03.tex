\begin{frame}[fragile]{Définition des modèles}
	\begin{columns}[T]
		\column{0.48\textwidth}
		
		\textbf{Définition des modèles}
		\begin{itemize}
			\item Plusieurs modèles de Machine Learning ont été implémentés selon une stratégie One-vs-Rest (OvR).
			\item Tous les modèles sont intégrés dans une pipeline comprenant : une normalisation \textbf{RobustScaler} et le classifieur OvR.
			\item Les classes sont pondérées afin de limiter l'impact du déséquilibre entre classes.
			\item Classificateurs testés : Logistic Regression, Linear SVM, Random Forest, HistGradientBoosting et SGD.
		\end{itemize}
		
		\column{0.48\textwidth}
		
		\begin{lstlisting}[style=terminal,language=python]
def get_ovr_hgb_model():
  scaler = RobustScaler()
  model = OneVsRestClassifier(
    HistGradientBoostingClassifier(
      learning_rate=0.1,
      max_depth=8,
      max_iter=200,
      class_weight="balanced",
      random_state=RANDOM_STATE,
    )
  )
  return Pipeline(
    steps=[
      ("scaler", scaler),
      ("model", model),
    ]
  )
\end{lstlisting}
	\end{columns}
\end{frame}

\begin{frame}{Métriques}
	\textbf{Métriques ML}
	\begin{itemize}
		\item \textbf{F1-micro} : Cette métrique réponds à la question "Quelle est la qualité globale de la transcription ?".
		\item \textbf{F1-macro} : Cette métrique donne le même poids aux notes fréquentes et rares. Elle permet d'évaluer la capacité du modèle à généraliser sur l'ensemble du registre.
		\item \textbf{Subset Accuracy} : Cette métrique réponds à la question "Combien de frames sont parfaitement transcrites ?".
	\end{itemize}
	
	\vspace{0.2cm}
	
	\textbf{Métrique musicales}
	\begin{itemize}
		\item \textbf{Hamming Loss} : Mesure le taux d'erreur moyen par note et par frame.
		\item \textbf{Pitch Tolerance Accuracy} : Cette métrique introduit une notion de proximité musicale.
		\item \textbf{Jitter} : Cette métrique mesure la stabilité de la transcription.
	\end{itemize}
\end{frame}
