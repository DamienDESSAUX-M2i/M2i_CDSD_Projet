\section{BC04 - Analyse prédictive de données non-structurées par l'intelligence artificielle}

\begin{frame}
    \centering
    \huge
    BC04 - Analyse prédictive de données non-structurées par l'intelligence artificielle
\end{frame}

\begin{frame}{Métriques}
	\textbf{Métriques ML}
	\begin{itemize}
		\item \textbf{F1-micro} : Quelle est la qualité globale de la transcription ?
		\item \textbf{F1-macro} : Permet d'évaluer la capacité du modèle à généraliser sur l'ensemble du registre.
		\item \textbf{Subset Accuracy} : Combien de frames sont parfaitement transcrites ?
	\end{itemize}
	
	\vspace{0.2cm}
	
	\textbf{Métrique musicales}
	\begin{itemize}
		\item \textbf{Hamming Loss} : Taux d'erreur moyen par note et par frame.
		\item \textbf{Pitch Tolerance Accuracy} : Notion de proximité musicale.
		\item \textbf{Jitter} : Stabilité de la transcription.
	\end{itemize}
\end{frame}

		\draw[arrow] (mlp) -- (output);
		
	\end{tikzpicture}
	
	\vspace{0.3cm}
	
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.48\textwidth}
			\textbf{Apport par rapport au MLP}
			\begin{itemize}
				\item Exploitation de la structure temps--fréquence de la CQT.
				\item Convolutions à différentes échelles.
				\item Pooling pour réduire la dimension.
				\item Global Average Pooling avant le MLP.
			\end{itemize}
		\end{column}
		
		\begin{column}{0.48\textwidth}
			\textbf{Variantes expérimentées}
			\begin{itemize}
				\item \textbf{Light} : coût de calcul réduit
				\item \textbf{Balanced} : compromis capacité / coût
				\item \textbf{Frequency} : contexte harmonique élargi
			\end{itemize}
		\end{column}
	\end{columns}
	
	\vspace{0.2cm}
	
	\begin{block}{Objectif expérimental}
		Étudier l'influence de l'exploitation des motifs locaux temps-fréquence.
	\end{block}
\end{frame}
