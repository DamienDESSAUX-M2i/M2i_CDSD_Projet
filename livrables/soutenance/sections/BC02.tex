\begin{frame}{Présentation des datasets}
    \textbf{Objectif de l'analyse}
    \begin{itemize}
        \item Étudier la structure et la qualité des données audio de guitare
        \item Évaluer leur pertinence pour une tâche de transcription automatique vers MIDI
    \end{itemize}

    \vspace{0.4cm}

    \textbf{Datasets utilisés}

    \begin{itemize}
        \item \textbf{GuitarSet \cite{xi2018guitarset}}
        \begin{itemize}
            \item Enregistrements de guitare solo (audio + annotations)
            \item Données alignées temporellement (value, time, duration)
            \item Large couverture de styles de jeu
        \end{itemize}

        \item \textbf{IDMT-SMT-Guitar \cite{kehling2023idmt}}
        \begin{itemize}
            \item Dataset orienté reconnaissance musicale
            \item Contient audio et annotations MIDI associées
            \item Variété de conditions d'enregistrement
        \end{itemize}
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Enjeu métier}
    \begin{itemize}
        \item Vérifier la compatibilité des données avec un pipeline de transcription audio $\rightarrow$ MIDI
    \end{itemize}
\end{frame}

% ===

\begin{frame}{Analyse du dataset GuitarSet}
    \textbf{Objectif de l'analyse}
    \begin{itemize}
        \item Comprendre la structure et la richesse du dataset
        \item Évaluer sa pertinence pour la transcription audio $\rightarrow$ MIDI
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Axes d'analyse}

    \begin{itemize}
        \item Structure des données (audio / annotations / alignement)
        \item Analyse musicale descriptive (pitch, densité, accords)
        \item Analyse du signal audio (spectrogrammes, dynamique)
        \item Qualité des annotations temporelles
    \end{itemize}

    \vspace{0.3cm}

    \textbf{Conclusion intermédiaire}
    \begin{itemize}
        \item Dataset riche mais partiellement contraint
        \item Bonne base pour apprentissage supervisé
        \item Complément nécessaire avec IDMT-SMT-Guitar
    \end{itemize}
\end{frame}
