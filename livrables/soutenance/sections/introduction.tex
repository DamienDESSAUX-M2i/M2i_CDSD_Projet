\section{Introduction}

\begin{frame}{Introduction du projet}

\textbf{Contexte métier}
\begin{itemize}
    \item La retranscription manuelle d'un audio de guitare vers un fichier MIDI est une tâche chronophage et experte.
    \item Des solutions existantes cependant la transcription fiable de guitare polyphonique dans des conditions réelles demeure un sujet complexe.
    \item Besoin d'outils d'assistance à la création musicale et pédagogique
\end{itemize}

\vspace{0.3cm}

\textbf{Projet GuitarFlow}
\begin{itemize}
    \item Prototype de transcription automatique d'un audio de guitare vers un fichier MIDI.
    \item Génération d'un fichier MIDI exploitable en MAO.
    \item Visualisation piano-roll et interface web de démonstration.
\end{itemize}

\vspace{0.3cm}

\textbf{Objectif du POC}
\begin{itemize}
    \item Valider la faisabilité technique d'un pipeline IA audio $\rightarrow$ MIDI.
    \item Évaluer les performances du système sur des données réelles.
    \item Préparer une future phase d'industrialisation.
\end{itemize}

\end{frame}
